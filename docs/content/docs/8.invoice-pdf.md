This guide builds a complete invoice. At the end, `GET /api/invoices/INV-2048`
returns a one-page PDF with the seller, the customer, the line items, tax, the
total, and a page number. The example uses only built-in fonts, so it needs no
font files.

![The invoice that this guide builds](/showcase/guide-invoice.webp)

Install the module first. See [Install Nuxt PDF]($docs.installation).

## Define the invoice data

Put the invoice type, a money formatter, and sample data in one file. The
template and your server code both import it, so they cannot disagree about
the shape of an invoice.

```ts [shared/invoice.ts]
export interface InvoiceLine {
  id: string
  description: string
  quantity: number
  unitPrice: number
}

export interface Invoice {
  number: string
  issueDate: string
  dueDate: string
  currency: string
  seller: { name: string, address: string[] }
  customer: { name: string, address: string[] }
  lines: InvoiceLine[]
  taxRate: number
}

export const formatMoney = (value: number, currency: string) =>
  new Intl.NumberFormat('en-US', { style: 'currency', currency }).format(value)

export const sampleInvoice: Invoice = {
  number: 'INV-2048',
  issueDate: 'March 3, 2026',
  dueDate: 'April 2, 2026',
  currency: 'USD',
  seller: { name: 'Acme Studio', address: ['500 Market Street', 'San Francisco, CA 94105'] },
  customer: { name: 'Northwind Traders', address: ['20 Harbor Road', 'Portland, OR 97201'] },
  lines: [
    { id: 'design', description: 'Website design', quantity: 1, unitPrice: 4800 },
    { id: 'dev', description: 'Frontend development (hours)', quantity: 32, unitPrice: 120 },
    { id: 'hosting', description: 'Hosting, 12 months', quantity: 1, unitPrice: 360 },
  ],
  taxRate: 0.08,
}
```

Use the currency code from your data. `Intl.NumberFormat` then prints `$`, `€`,
or `£` with the correct separators.

## Write the template

Create `pdfs/invoice.vue`. The module turns every file in `pdfs/` into a
template, so this one becomes `pdf.invoice`.

```vue [pdfs/invoice.vue]
<script setup lang="ts">
import type { Invoice } from '../shared/invoice'
import { formatMoney, sampleInvoice } from '../shared/invoice'

const props = defineProps<{ invoice: Invoice }>()

definePdf<{ invoice: Invoice }>({
  title: ({ invoice }) => `Invoice ${invoice.number}`,
  filename: ({ invoice }) => `invoice-${invoice.number}.pdf`,
  language: 'en-US',
  sampleData: { invoice: sampleInvoice },
})

const invoice = props.invoice
const money = (value: number) => formatMoney(value, invoice.currency)
const subtotal = invoice.lines.reduce((sum, line) => sum + line.quantity * line.unitPrice, 0)
const tax = subtotal * invoice.taxRate
const total = subtotal + tax

const muted = '#64748B'
const label = { color: muted, fontSize: 8, marginBottom: 4 }
const columns = { description: 260, quantity: 50, price: 90, amount: 99 }
</script>

<template>
  <PdfDocument>
    <PdfPage size="A4" :style="{ color: '#0F172A', fontSize: 10, padding: 48, paddingBottom: 72 }">
      <PdfView :style="{ flexDirection: 'row', justifyContent: 'space-between' }">
        <PdfView>
          <PdfText :style="{ fontSize: 14, fontWeight: 700 }">{{ invoice.seller.name }}</PdfText>
          <PdfText v-for="line in invoice.seller.address" :key="line" :style="{ color: muted }">
            {{ line }}
          </PdfText>
        </PdfView>
        <PdfView :style="{ alignItems: 'flex-end' }">
          <PdfText :style="{ fontSize: 24, fontWeight: 700 }">Invoice</PdfText>
          <PdfText :style="{ color: muted }">{{ invoice.number }}</PdfText>
        </PdfView>
      </PdfView>

      <PdfView :style="{ flexDirection: 'row', marginTop: 40 }">
        <PdfView :style="{ width: '50%' }">
          <PdfText :style="label">Bill to</PdfText>
          <PdfText :style="{ fontWeight: 700 }">{{ invoice.customer.name }}</PdfText>
          <PdfText v-for="line in invoice.customer.address" :key="line">{{ line }}</PdfText>
        </PdfView>
        <PdfView :style="{ width: '25%' }">
          <PdfText :style="label">Issued</PdfText>
          <PdfText>{{ invoice.issueDate }}</PdfText>
        </PdfView>
        <PdfView :style="{ width: '25%' }">
          <PdfText :style="label">Due</PdfText>
          <PdfText>{{ invoice.dueDate }}</PdfText>
        </PdfView>
      </PdfView>

      <PdfView :style="{ borderBottomColor: '#0F172A', borderBottomWidth: 1, flexDirection: 'row', marginTop: 32, paddingBottom: 6 }">
        <PdfText :style="{ ...label, marginBottom: 0, width: columns.description }">Description</PdfText>
        <PdfText :style="{ ...label, marginBottom: 0, textAlign: 'right', width: columns.quantity }">Qty</PdfText>
        <PdfText :style="{ ...label, marginBottom: 0, textAlign: 'right', width: columns.price }">Price</PdfText>
        <PdfText :style="{ ...label, marginBottom: 0, textAlign: 'right', width: columns.amount }">Amount</PdfText>
      </PdfView>
      <PdfView
        v-for="line in invoice.lines"
        :key="line.id"
        :wrap="false"
        :style="{ borderBottomColor: '#E2E8F0', borderBottomWidth: 1, flexDirection: 'row', paddingVertical: 8 }"
      >
        <PdfText :style="{ width: columns.description }">{{ line.description }}</PdfText>
        <PdfText :style="{ textAlign: 'right', width: columns.quantity }">{{ line.quantity }}</PdfText>
        <PdfText :style="{ textAlign: 'right', width: columns.price }">{{ money(line.unitPrice) }}</PdfText>
        <PdfText :style="{ textAlign: 'right', width: columns.amount }">{{ money(line.quantity * line.unitPrice) }}</PdfText>
      </PdfView>

      <PdfView :wrap="false" :style="{ marginLeft: 'auto', marginTop: 16, width: 220 }">
        <PdfView :style="{ flexDirection: 'row', justifyContent: 'space-between', paddingVertical: 3 }">
          <PdfText :style="{ color: muted }">Subtotal</PdfText>
          <PdfText>{{ money(subtotal) }}</PdfText>
        </PdfView>
        <PdfView :style="{ flexDirection: 'row', justifyContent: 'space-between', paddingVertical: 3 }">
          <PdfText :style="{ color: muted }">Tax {{ Number((invoice.taxRate * 100).toFixed(4)) }}%</PdfText>
          <PdfText>{{ money(tax) }}</PdfText>
        </PdfView>
        <PdfView :style="{ backgroundColor: '#0F172A', borderRadius: 4, flexDirection: 'row', justifyContent: 'space-between', marginTop: 8, padding: 10 }">
          <PdfText :style="{ color: '#FFFFFF', fontWeight: 700 }">Total due</PdfText>
          <PdfText :style="{ color: '#FFFFFF', fontWeight: 700 }">{{ money(total) }}</PdfText>
        </PdfView>
      </PdfView>

      <PdfText
        fixed
        :render="({ pageNumber, totalPages }) => `${invoice.number} · Page ${pageNumber} of ${totalPages}`"
        :style="{ bottom: 32, color: muted, fontSize: 8, left: 48, position: 'absolute', right: 48, textAlign: 'right' }"
      />
    </PdfPage>
  </PdfDocument>
</template>
```

Four details keep the invoice correct when it grows:

- `wrap="false"` on each line moves the whole row to the next page. A page
  break never separates a description from its amount.
- `wrap="false"` on the totals block keeps subtotal, tax, and total together.
- The `fixed` footer repeats on every page. `paddingBottom: 72` on the page
  keeps the line items clear of it.
- `columns` holds the column widths once. The header and the rows read the
  same values, so they stay aligned.

## Check the layout in the preview

Start the development server and open `http://localhost:3000/_pdf/invoice`. The
preview renders the template with `sampleData`. It updates when you save the
file.

```bash
pnpm dev
```

The preview and `sampleData` exist only in development. A production build
removes both.

## Serve the PDF from a route

Create `server/api/invoices/[id].get.ts`. The route loads the invoice and
returns the PDF. `findInvoice` stands for your own function in `server/utils/`.
It must load the invoice and check that the current user may see it.

```ts
import { pdf } from '#pdf'

export default defineEventHandler(async (event) => {
  const id = getRouterParam(event, 'id')
  const invoice = id ? await findInvoice(event, id) : undefined
  if (!invoice) {
    throw createError({ statusCode: 404, statusMessage: 'Invoice not found' })
  }

  const result = await pdf.invoice.render({ invoice })
  return result.response({ headers: { 'cache-control': 'private, no-store' } })
})
```

TypeScript checks the props of `pdf.invoice.render()` against the template. A
missing or misspelled field shows in your editor, not in production.

`response()` sends the file as a download named `invoice-INV-2048.pdf`, the
`filename` from `definePdf`. To open the PDF in the browser tab instead, pass
`disposition: 'inline'`:

```ts
result.response({ disposition: 'inline' })
```

Link to the route from a page such as `app/pages/invoices/[id].vue`. Use a
plain link, so the browser handles the download:

```vue
<script setup lang="ts">
const route = useRoute()
</script>

<template>
  <a :href="`/api/invoices/${route.params.id}`">Download PDF</a>
</template>
```

## Attach the PDF to an email

Use the bytes when the PDF does not go to the browser. `toBuffer()` returns a
Node `Buffer` that most email and storage clients accept:

```ts
const result = await pdf.invoice.render({ invoice })
const content = await result.toBuffer()
const filename = result.metadata.filename ?? 'invoice.pdf'
```

Pass `content` and `filename` to your email provider's attachment field, or
upload `content` to object storage.

## Test the invoice

Render the real template in Vitest and check the text in the PDF. Install the
test dependencies first, as described in [Test a PDF]($docs.testing).

```ts [test/invoice.test.ts]
import { expectPdf, renderPdfSfc } from '@lupinum/nuxt-pdf/test'
import { it } from 'vitest'
import { sampleInvoice } from '../shared/invoice'

it('prints the invoice number and the total', async () => {
  const { parsed } = await renderPdfSfc('./pdfs/invoice.vue', { invoice: sampleInvoice })

  expectPdf(parsed)
    .toHavePageCount(1)
    .toContainText('INV-2048')
    .toContainText('$9,720.00')
})
```

The test fails when a change drops the total or pushes the invoice onto a
second page.

## Use your brand fonts and logo

Put font files in `pdfs/fonts/` and images in `pdfs/assets/`, then register the
fonts in `nuxt.config.ts`. The module checks and bundles both at build time. See
[Add images and fonts]($docs.images-and-fonts) for the configuration, and the
[production recipes]($docs.recipes) for payment details, QR codes, and long
invoices with continuation headers.
